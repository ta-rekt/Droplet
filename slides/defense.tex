		P_{\rm poly}(d) \propto d^2e^{-3d^2/2sa^2} \propto e^{-U_{\rm poly}(d)/k_BT}

	U_{\rm tot}(d) = U_{\rm attr}(d) + U_{\rm poly}(d)

	P_{\rm eq}(d) \propto e^{-U_{\rm tot}(d)/k_BT}
